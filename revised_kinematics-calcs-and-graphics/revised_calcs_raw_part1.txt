alpha - first servo angle
gamma - second servo angle

a = 61
b = 71
c = 150

D = sqrt(h^2 + r^2)

beta = acos((c^2 + c^2 - D^2) / (2 * c^2)) = acos(45000 - D^2 / 45000)

alpha = arctan(h / r) + acos((c^2 + d^2 - c^2) / (2 * c * d)) 
alpha = arctan(h / r) + acos(d / 300)

d^2 = b^2 + c^2 - 2 * b * c * cos(180 - beta)

d = sqrt(71^2 + 150^2 + 2 * 71 * 150 * cos(beta)) = sqrt(27541 + 21300 * cos(beta))

gamma = acos((c^2 + d^2 - b^2) / (2 * c * d)) + acos((a^2 + d^2 - c^2) / (2 * a * d)) - pi/2 + alpha 

gamma = acos((17459 + d^2) / (300 * d)) + acos((d^2 - 18779) / (122 * d)) + alpha - pi/2
